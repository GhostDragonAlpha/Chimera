# star_system.chimera -- Phase 0
physics kepler {
  body star:    mass=1.989e30, pos=(0,0,0)
  body earth:   mass=5.972e24, pos=(1.5e11,0,0), vel=(0,29780,0)
  body mars:    mass=6.39e23, pos=(2.28e11,0,0), vel=(0,24077,0)
  force gravity { G = 6.674e-11; F=G*m1*m2/r^2 }
  integrator symplectic_euler { dt=60; v+=a*dt; p+=v*dt }
}
